import json,glob,os
out=open('refsummary.txt','w',encoding='utf-8')
for f in sorted(os.listdir('references')):
    r=json.load(open('references/'+f,encoding='utf-8'))
    d=dict(r['digest'])
    for k in ['next','budget_left','cached','not_shown','judgement']: d.pop(k,None)
    out.write('=== '+f+'\n')
    for k in ['fen_before','fen_after','move','uci','moves_before']: out.write(f'{k}: {r[k]}\n')
    out.write('KEY_CAUSES:\n')
    for i,c in enumerate(r['key_causes'],1): out.write(f'  {i} [{c["role"]}/{c["kind"]}] {c["text"]}\n')
    out.write('DIGEST: '+json.dumps(d,ensure_ascii=False,indent=1)+'\n')
    out.write('FULLTEXT: '+json.dumps(r['full'].get('text'),ensure_ascii=False)+'\n\n')
b=json.load(open('bundle.json',encoding='utf-8'))
out2=open('items.txt','w',encoding='utf-8')
items=sorted(b['items'],key=lambda x:x['moment'])
for it in items:
    out2.write(f"### {it['key']} {it['moment']}\n")
    for s in it['sentences']: out2.write(f"  {s['n']} ({s['class'][0]}) {s['text']}\n")
print(len(items))
from collections import Counter
print(Counter(i['moment'] for i in items))
EOF
